cmake_minimum_required(VERSION 3.24)
project(ExternalCLabConsumer LANGUAGES CXX)
find_package(CLab CONFIG REQUIRED)
if(TARGET clab::substrate_api)
    message(FATAL_ERROR "Evaluator-only install exported a substrate target")
endif()
foreach(library IN ITEMS foundation observation trace evaluator)
    get_target_property(dependencies clab::${library} INTERFACE_LINK_LIBRARIES)
    if(NOT dependencies)
        set(dependencies "")
    endif()
    set(expected "")
    if(library STREQUAL "observation")
        set(expected "clab::foundation")
    elseif(library STREQUAL "evaluator" OR library STREQUAL "trace")
        set(expected "clab::observation")
    endif()
    if(NOT dependencies STREQUAL expected)
        message(FATAL_ERROR "Unexpected exported dependencies for clab::${library}: ${dependencies}")
    endif()
endforeach()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE clab::trace clab::evaluator)
